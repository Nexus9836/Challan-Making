    <script>
        const STORAGE_KEY = 'challan-data';

        function safeExecute(callback, fallbackValue) {
            try {
                return callback();
            } catch (error) {
                console.error('Safe execution error:', error);
                return fallbackValue;
            }
        }

        function safeNumber(value, fallback = 0) {
            const parsed = Number(value);
            return Number.isFinite(parsed) ? parsed : fallback;
        }

        function getElement(id) {
            return document.getElementById(id);
        }

        document.addEventListener('DOMContentLoaded', function () {
            safeExecute(function () {
                loadData();
            }, null);
        });

        function saveData() {
            safeExecute(function () {
                if (!window.localStorage) return;
                const data = getFormData();
                window.localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
            }, null);
        }

        function loadData() {
            safeExecute(function () {
                if (!window.localStorage) return;
                const raw = window.localStorage.getItem(STORAGE_KEY);
                if (!raw) return;
                const parsed = JSON.parse(raw);
                setFormData(parsed);
            }, null);
        }

        function addRow() {
            safeExecute(function () {
                const tbody = document.getElementById('itemRows');
                if (!tbody) return;
                const template = tbody.rows[0];
                if (!template) return;

                const newRow = template.cloneNode(true);
                const deleteBtn = newRow.querySelector('button');
                if (deleteBtn) {
                    deleteBtn.onclick = function () { deleteRow(this); };
                }

                tbody.appendChild(newRow);
                calculateTotals();
                saveData();
            }, null);
        }

        function deleteRow(btn) {
            safeExecute(function () {
                const row = btn && btn.closest('tr');
                if (!row) return;
                const tbody = row.parentNode;
                if (!tbody) return;
                if (tbody.rows.length > 1) {
                    row.remove();
                    calculateTotals();
                    saveData();
                }
            }, null);
        }

        function calculateTotals() {
            safeExecute(function () {
                let total = 0;
                const rows = document.querySelectorAll('#itemRows tr');

                rows.forEach(function (row) {
                    const inputs = row.querySelectorAll('input');
                    if (!inputs || inputs.length < 5) return;

                    const bodyValue = safeNumber(inputs[4] ? inputs[4].value : 0, 0);
                    total += bodyValue;
                });

                const totalElement = getElement('footerTotal');
                if (totalElement) totalElement.textContent = total.toFixed(3);

                const bConsElement = getElement('footerBCons');
                if (bConsElement) bConsElement.textContent = (total * safeNumber(getElement('consumptionInput')?.value || 0, 0)).toFixed(5);

                const tConsElement = getElement('footerTCons');
                if (tConsElement) tConsElement.textContent = total.toFixed(5);

                const headerCalc = getElement('headerCalculatedCons');
                if (headerCalc) {
                    const consumption = safeNumber(getElement('consumptionInput')?.value || 0, 0);
                    headerCalc.textContent = (total * consumption).toFixed(5);
                }

                saveData();
            }, null);
        }

        function triggerImport() {
            safeExecute(function () {
                const input = getElement('importFile');
                if (input) input.click();
            }, null);
        }

        function importData(event) {
            safeExecute(function () {
                const file = event && event.target ? event.target.files[0] : null;
                if (!file) return;

                const reader = new FileReader();
                reader.onload = function (e) {
                    try {
                        const text = e && e.target ? e.target.result : null;
                        if (!text) {
                            alert('File is empty.');
                            return;
                        }
                        const parsed = JSON.parse(text);
                        setFormData(parsed);
                    } catch (error) {
                        console.error('Import parse error:', error);
                        alert('Invalid file format');
                    }
                };
                reader.readAsText(file);
            }, null);
        }

        function exportData() {
            safeExecute(function () {
                const data = getFormData();
                const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' });
                const url = URL.createObjectURL(blob);
                const link = document.createElement('a');
                link.href = url;
                link.download = 'challan-export.json';
                document.body.appendChild(link);
                link.click();
                document.body.removeChild(link);
                URL.revokeObjectURL(url);
            }, null);
        }

        }

        function resetAllData() {
            safeExecute(function () {
                if (!confirm('Reset all data?')) return;
                if (window.localStorage) {
                    window.localStorage.removeItem(STORAGE_KEY);
                }
                window.location.reload();
            }, null);
        }

        function getFormData() {
            return safeExecute(function () {
                return {
                    quality: getElement('globalQuality') ? getElement('globalQuality').value : '',
                    remarks: getElement('qualityRemarksInput') ? getElement('qualityRemarksInput').value : '',
                    consumption: getElement('consumptionInput') ? getElement('consumptionInput').value : '',
                    gsm: getElement('gsmInput') ? getElement('gsmInput').value : '',
                    fabricGsm: getElement('fabricGsmInput') ? getElement('fabricGsmInput').value : '',
                    timestamp: new Date().toISOString()
                };
            }, {});
        }

        function setFormData(data) {
            safeExecute(function () {
                if (!data || typeof data !== 'object') return;

                if (data.quality && getElement('globalQuality')) {
                    getElement('globalQuality').value = data.quality;
                }

                if (data.remarks && getElement('qualityRemarksInput')) {
                    getElement('qualityRemarksInput').value = data.remarks;
                }

                if (data.consumption && getElement('consumptionInput')) {
                    getElement('consumptionInput').value = data.consumption;
                }

                if (data.gsm && getElement('gsmInput')) {
                    getElement('gsmInput').value = data.gsm;
                }

                if (data.fabricGsm && getElement('fabricGsmInput')) {
                    getElement('fabricGsmInput').value = data.fabricGsm;
                }

                calculateTotals();
            }, null);
        }
